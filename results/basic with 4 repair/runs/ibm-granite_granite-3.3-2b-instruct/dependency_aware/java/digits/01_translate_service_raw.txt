public class Service {
    public static int score_digits(int n) {
        return digit_sum(n) * 3;
    }

    private static int digit_sum(int n) {
        n = Math.abs(n);
        int total = 0;
        while (n > 0) {
            total += n % 10;
            n /= 10;
        }
        return total;
    }
}
