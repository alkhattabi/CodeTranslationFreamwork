public class Main {
    public static int solve(int n) {
        int value = scoreDigits(n);
        if (value % 2 == 0) {
            return value + 1;
        } else {
            return value - 1;
        }
    }

    // Assuming Service.java has scoreDigits method
    public static int scoreDigits(int n) {
        return digitSum(n) * 3;
    }

    private static int digitSum(int n) {
        int sum = 0;
        while (n!= 0) {
            sum += n % 10;
            n /= 10;
        }
        return sum;
    }
}
