```javascript
// utils.mjs
export function tier(n) {
    if (n >= 100) {
        return 3;
    }
    if (n >= 50) {
        return 2;
    }
    if (n >= 10) {
        return 1;
    }
    return 0;
}
```

```javascript
// service.mjs
import { tier } from "./utils.mjs";

export function discount(n) {
    return tier(n) * 5;
}
```

Please note that the `discount` function in the `service.mjs` file is using the `tier` function from the `utils.mjs` module.
