#include "api.hpp"

int matrix_energy(int n) {
    int x = std::max(-100, std::min(100, n));

    int m[3][3] = {
        {std::max(-100, std::min(100, x)) % 5, 2, -1},
        {3, std::max(-100, std::min(100, x)) % 7, 4},
        {1, std::max(-100, std::min(100, x)) % 3, 0}
    };

    int square[3][3] = {
        {m[0][0] % 5, m[0][1], m[0][2]},
        {m[1][0], m[1][1] % 7, m[1][2]},
        {m[2][0], m[2][1], m[2][2]}
    };

    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            square[i][j] = (i == j)? square[i][j] : (square[0][0] * square[i][j] + square[j][0] * square[i][j]);
        }
    }

    int diagonal = 0;
    for (int i = 0; i < 3; ++i) {
        diagonal += square[i][i];
    }

    return diagonal + vector_signature(n);
}
