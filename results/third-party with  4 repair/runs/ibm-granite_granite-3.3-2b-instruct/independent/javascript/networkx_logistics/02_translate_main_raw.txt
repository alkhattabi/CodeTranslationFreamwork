```javascript
import { disruption_score } from "./service.mjs";
import { route_cost } from "./utils.mjs";

export function solve(n) {
    const graph = new require("networkx").Graph();
    const offset = Math.abs(n) % 7;
    graph.setWeightedEdgesFrom([
        [0, 1, 3 + offset],
        [0, 2, 5],
        [1, 2, 1 + offset % 3],
        [1, 3, 4],
        [2, 3, 2 + offset % 2],
        [3, 4, 6],
        [2, 4, 8 + offset]
    ]);
    const tree = graph.minimumSpanningTree({ weight: "weight" });
    const total = tree.size({ weight: "weight" }).toString();
    return parseInt(total) + disruption_score(n) - route_cost(n);
}
```
Note: The "networkx" library in JavaScript does not exist as a built-in library. You would need to use a JavaScript library like "networkx-js" or implement the graph algorithms manually. The provided code assumes the existence of such a library for the purpose of this translation.
