```javascript
// main.mjs
import { renewal_date } from "./utils.mjs";
import { grace_period } from "./service.mjs";

export function solve(n) {
    const renewed = renewal_date(n);
    const deadline = grace_period(n);
    if (renewed < 0 || deadline < 0) {
        return -1;
    }
    const y1 = renewed / 10000;
    const m1 = (renewed / 100) % 100;
    const d1 = renewed % 100;
    const y2 = deadline / 10000;
    const m2 = (deadline / 100) % 100;
    const d2 = deadline % 100;
    const date1 = new Date(y1 * 10000 + m1 * 100 + d1);
    const date2 = new Date(y2 * 10000 + m2 * 100 + d2);
    const diffInDays = Math.floor((date2 - date1) / (1000 * 60 * 60 * 24));
    return diffInDays * 100 + d2;
}
```

Note: JavaScript's Date object does not directly support subtraction to find the difference in days between two dates. Therefore, we need to manually calculate the difference in milliseconds, convert it to days, and then multiply by 100 to get the final result as per the Python implementation.
