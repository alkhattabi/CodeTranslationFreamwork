            }

    }

    private static int divisorCount(int value) {
        int count = 0;
        for (int i = 2; i <= Math.sqrt(value); i++) {
            while (value % i == 0) {
                count++;
                value /= i;
            }
        }
        if (value > 1) {
            count++;
        }
        return count;
    }
