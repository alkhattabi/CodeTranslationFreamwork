import sympy as sp

def factor_signature(n: int) -> int:
    value = abs(n) % 1500 + 2
    factors = sp.factorint(value)
    score = sum(int(prime) * int(exponent) for prime, exponent in factors.items())
    return score + 3 * len(factors)
