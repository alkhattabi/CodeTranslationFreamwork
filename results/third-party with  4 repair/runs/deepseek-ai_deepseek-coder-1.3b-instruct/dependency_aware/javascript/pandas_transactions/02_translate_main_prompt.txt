def solve(n: int) -> int:

    alert = rolling_alert(n)
    scores = pd.Series([abs(balance) % 50, alert, abs(n) % 50])

export function rolling_alert(n) {
    n = Math.max(-200, Math.min(200, n));
    let series = [n % 13, (n // 2) % 13, n % 17, Math.abs(n) % 11, n % 7];
    let rolling = series.reduce((a, b) => a + b, 0);
    let alerts = Math.floor(rolling / 15) * 10 + Math.abs(category_balance(n)) % 23;
    return alerts;
}

export function category_balance(n) {
    n = Math.max(-200, Math.min(200, n));
    let frame = {
        "category": ["A", "B", "A", "C", "B", "C", "A"],
        "amount": [n, 12, Math.floor(n / 2), -7, Math.floor(n / 9), 4, -Math.floor(n / 3)]
    };
    let sums = frame.amount.reduce((a, b) => a + b, 0);
    return 2 * sums - sums.filter(x => x < 0).reduce((a, b) => a + b, 0);
}
