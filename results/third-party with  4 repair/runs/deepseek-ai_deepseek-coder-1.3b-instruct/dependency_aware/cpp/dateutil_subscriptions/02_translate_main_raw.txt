Here is the C++ implementation of the Python code. Note that the C++ standard library does not have a direct equivalent to Python's datetime module, so we use the <chrono> library for date handling.

```cpp
#include "api.hpp"
#include <chrono>
#include <stdexcept>

int solve(int n) {
    // Implementation of solve function
    return n * n;
}

int grace_period(int n) {
    // Implementation of grace_period function
    return n / 2;
}

int renewal_date(int n) {
    if (n < 19000101 || n > 20991231) {
        return -1;
    }
    int year, month, day;
    year = n / 10000;
    month = (n / 100) % 100;
    day = n % 100;
    if (year < 0 || month < 0 || day < 0) {
        return -1;
    }
    try {
        std::chrono::year y(year);
        std::chrono::month m(month);
        std::chrono::day d(day);
        std::chrono::time_point<std::chrono::system_clock> start = std::chrono::year_month_day{y, m, d};
        std::chrono::time_point<std::chrono::system_clock> end = start + std::chrono::months(1);
        return std::chrono::floor<int>(end.time_since_epoch() / std::chrono::seconds(1));
    } catch (std::exception& e) {
        return -1;
    }
}
```

This code assumes that the `renewal_date` function returns the number of days since the start of the year, rounded down to the nearest whole day. If the `renewal_date` function throws an exception, it returns -1.
