import pandas as pd
from utils import category_balance
from service import rolling_alert

def solve(n: int) -> int:
    balance = category_balance(n)
    alert = rolling_alert(n)
    scores = pd.Series([abs(balance) % 50, alert, abs(n) % 50])
    ranks = scores.rank(method="dense", ascending=False)
    penalty = int(ranks.iloc[0] * 7 + ranks.iloc[1] * 3)
    return balance + alert - penalty

import { category_balance } from "./utils.mjs";

export function rolling_alert(n) {
    n = Math.max(-200, Math.min(200, n));
    const series = new Array(5).fill(n % 13);
    series[1] = (n >>> 1) % 13;
    series[2] = n % 17;
    series[3] = Math.abs(n) % 11;
    series[4] = n % 7;
    const rolling = series.reduce((acc, val) => acc + val, 0);
    const alerts = (rolling >= 15).reduce((acc, val) => acc + val, 0);
    return alerts * 10 + category_balance(n) % 23;
}

export function category_balance(n) {
    n = Math.max(-200, Math.min(200, n));
    const frame = new DataFrame({
        "category": ["A", "B", "A", "C", "B", "C", "A"],
        "amount": [n, 12, Math.floor(n / 2), -7, n % 9, 4, -Math.floor(n / 3)]
    });
    const sums = frame.groupby("category", sort=true).sum();
    return parseInt(2 * sums.get("A") + sums.get("B") - sums.get("C"));
}
