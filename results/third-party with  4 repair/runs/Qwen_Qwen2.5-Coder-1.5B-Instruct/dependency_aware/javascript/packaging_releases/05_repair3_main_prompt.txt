export function release_rank(n) {

        return -1;
    }
    const major = Math.floor(n / 10000);
    const minor = Math.floor((n / 100) % 100);
    const patch = n % 100;

        suffix = "rc1";
    }
    const version = `${major}.${minor}.${patch}${suffix}`;

    if (version.localeCompare(reference) < 0) {
        return 1;
