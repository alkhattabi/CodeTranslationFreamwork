int calculate_category_sum(const std::vector<int>& amounts) {
    int sum_A = 0;
    int sum_B = 0;
    int sum_C = 0;

    for (int amount : amounts) {
        if (amount >= 0) {
            if (amount % 3 == 0) {
                sum_A += amount;
            } else if (amount % 3 == 1) {
                sum_B += amount;
            } else {
                sum_C += amount;
            }
        } else {
            if (-amount % 3 == 0) {
                sum_A -= amount;
            } else if (-amount % 3 == 1) {
                sum_B -= amount;
            } else {
                sum_C -= amount;
            }
        }
    }

    return sum_A + sum_B - sum_C;
}

import pandas as pd

def category_balance(n: int) -> int:
    n = max(-200, min(200, n))
    frame = pd.DataFrame({
        "category": ["A", "B", "A", "C", "B", "C", "A"],
        "amount": [n, 12, n // 2, -7, n % 9, 4, -n // 3]
    })
    sums = frame.groupby("category", sort=True)["amount"].sum()
    return int(2 * sums.loc["A"] + sums.loc["B"] - sums.loc["C"])
