package com.example;

import org.apache.commons.math3.graph.DirectedWeightedGraph;
import org.apache.commons.math3.graph.SimpleDirectedWeightedGraph;
import org.apache.commons.math3.graph.indexed.IndexedWeightedGraph;
import org.apache.commons.math3.util.Pair;

import java.util.HashMap;
import java.util.Map;

public class Utils {

    public static int routeCost(int n) {
        int x = Math.abs(n) % 13;
        DirectedWeightedGraph<Integer, Integer> graph = new SimpleDirectedWeightedGraph<>(IndexedWeightedGraph::new);
        Map<Pair<Integer, Integer>, Double> edges = new HashMap<>();

        edges.put(new Pair<>(0, 1), 3 + x % 3);
        edges.put(new Pair<>(0, 2), 5 + x % 4);
        edges.put(new Pair<>(1, 3), 4 + x % 2);
        edges.put(new Pair<>(1, 4), 7);
        edges.put(new Pair<>(2, 3), 1 + x % 5);
        edges.put(new Pair<>(2, 4), 3 + x % 3);
        edges.put(new Pair<>(3, 5), 6);
        edges.put(new Pair<>(4, 5), 2 + x % 4);
        edges.put(new Pair<>(5, 6), 1 + x % 2);
        edges.put(new Pair<>(2, 6), 15);

        for (Map.Entry<Pair<Integer, Integer>, Double> entry : edges.entrySet()) {
            graph.addEdge(entry.getKey().getFirst(), entry.getKey().getSecond(), entry.getValue());
        }

        return (int) graph.getShortestPathDistance(0, 6, "weight");
    }
}
