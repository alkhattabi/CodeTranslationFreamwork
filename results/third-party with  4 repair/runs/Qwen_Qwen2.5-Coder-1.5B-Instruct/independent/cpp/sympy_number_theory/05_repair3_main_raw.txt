int number_strength(int n) {
    // Implement the logic for number strength calculation
    // Example: Calculate the sum of divisors
    int sum_of_divisors = 0;
    for (int i = 1; i <= n; ++i) {
        if (n % i == 0) {
            sum_of_divisors += i;
        }
    }
    return sum_of_divisors;
}

int factor_signature(int n) {
    // Implement the logic for factor signature calculation
    // Example: Calculate the product of all factors
    int product_of_factors = 1;
    for (int i = 1; i <= n; ++i) {
        if (n % i == 0) {
            product_of_factors *= i;
        }
    }
    return product_of_factors;
}
