def vector_signature(n: int) -> int:
    n = max(-100, min(100, n))
    v = np.array([n % 11, abs(n) % 7, (n // 3) % 9], dtype=np.int64)
    m = np.arange(1, 10, dtype=np.int64).reshape(3, 3)
    out = m @ v
