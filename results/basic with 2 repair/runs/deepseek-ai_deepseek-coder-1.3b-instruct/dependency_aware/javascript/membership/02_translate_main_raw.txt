// utils.mjs
export function tier(n) {
    if (n >= 100) {
        return 3;
    }
    if (n >= 50) {
        return 2;
    }
    if (n >= 10) {
        return 1;
    }
    return 0;
}

// service.mjs
export function discount(n) {
    return Math.floor(n / 10);
}

// main.mjs
import { discount } from "./service.mjs";
import { tier } from "./utils.mjs";

export function solve(n) {
    if (typeof n !== 'number' || n % 1 !== 0) {
        throw new Error('Input must be an integer');
    }
    let value = n * 4 - discount(n);
    value = Math.max(0, value);
    return value * tier(value);
}

The `discount` function is a dependency of `solve`, and the `tier` function is also a dependency of `solve`. The `solve` function also checks if the input is an integer and throws an error if it's not.
