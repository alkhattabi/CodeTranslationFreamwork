#pragma once
#include <algorithm>
#include <cmath>
#include <cstdlib>

// Helper function declarations from api.hpp
int solve(int n);
int discount(int n);

// Global function definitions
int tier(int n) {
    if (n >= 100) {
        return 3;
    }
    if (n >= 50) {
        return 2;
    }
    if (n >= 10) {
        return 1;
    }
    return 0;
}

int discount(int n) {
    return tier(n) * 5;
}
