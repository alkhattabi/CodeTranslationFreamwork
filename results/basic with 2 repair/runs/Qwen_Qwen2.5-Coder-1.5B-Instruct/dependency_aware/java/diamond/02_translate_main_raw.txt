import java.util.function.IntUnaryOperator;

public class Main {
    public static int solve(int n) {
        IntUnaryOperator normalized = Utils::normalize;
        IntUnaryOperator weighted = Service::weighted;
        return weighted.applyAsInt(n) + normalized.applyAsInt(n);
    }

    public static void main(String[] args) {
        // Example usage
        System.out.println(solve(5)); // Expected output: 13
        System.out.println(solve(-3)); // Expected output: 7
        System.out.println(solve(0)); // Expected output: 1
    }
}
