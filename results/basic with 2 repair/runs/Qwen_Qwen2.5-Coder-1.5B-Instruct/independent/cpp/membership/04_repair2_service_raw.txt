#pragma once
#include <algorithm>
#include <cmath>
#include <cstdlib>

// Helper function declaration from api.hpp
int solve(int n);

// Global function definition
int tier(int n) {
    if (n >= 100) {
        return 3;
    }
    if (n >= 50) {
        return 2;
    }
    if (n >= 10) {
        return 1;
    }
    return 0;
}

// Service class implementation
class Service {
public:
    // Method to calculate discount
    static int discount(int n) {
        return tier(n) * 5;
    }
};
