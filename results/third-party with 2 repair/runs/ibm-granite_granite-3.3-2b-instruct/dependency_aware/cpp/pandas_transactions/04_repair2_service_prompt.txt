import pandas as pd
from utils import category_balance

def rolling_alert(n: int) -> int:
    n = max(-200, min(200, n))
    series = pd.Series([n % 13, (n // 2) % 13, n % 17, abs(n) % 11, n % 7])
    rolling = series.rolling(window=3, min_periods=1).sum()
    alerts = int((rolling >= 15).sum())
    return alerts * 10 + abs(category_balance(n)) % 23

#include "api.hpp"

int category_balance(int n) {
    n = std::max(-200, std::min(200, n));

    std::vector<std::pair<std::string, int>> frame = {
        {"A", n},
        {"B", 12},
        {"A", n / 2},
        {"C", -7},
        {"B", n % 9},
        {"C", 4},
        {"A", -n / 3}
    };

    std::map<std::string, int> sums;
    for (const auto& pair : frame) {
        sums[pair.first] += pair.second;
    }

    return 2 * sums["A"] + sums["B"] - sums["C"];
}
