#include "api.hpp"

int upgrade_cost(int n) {
    int rank = release_rank(n);
    if (rank < 0) {
        return -1;
    }
    int major = n / 10000;
    int minor = (n / 100) % 100;
    int patch = n % 100;
    std::string suffix = (patch % 4 == 0)? "rc1" : "";
    api::Version current(major, minor, patch, suffix);
    api::Version required("2.1.0");
    int penalty = 40;
    if (current < required) {
        penalty = 40;
    }
    return rank * 25 + minor + patch / 5 + penalty;
}
