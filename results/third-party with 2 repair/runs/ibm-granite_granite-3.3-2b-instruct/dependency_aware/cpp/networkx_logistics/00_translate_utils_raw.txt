#include "api.hpp"

int route_cost(int n) {
    int x = abs(n) % 13;
    api::DiGraph graph;
    graph.add_weighted_edges_from({
        {0, 1, 3 + x % 3},
        {0, 2, 5 + x % 4},
        {1, 3, 4 + x % 2},
        {1, 4, 7},
        {2, 3, 1 + x % 5},
        {2, 4, 3 + x % 3},
        {3, 5, 6},
        {4, 5, 2 + x % 4},
        {5, 6, 1 + x % 2},
        {2, 6, 15}
    });
    return graph.shortest_path_length(0, 6, weight="weight");
}

Note: This code assumes that `api.hpp` is a header file containing the definition of `api::DiGraph` class with `add_weighted_edges_from` and `shortest_path_length` methods. The behavior of these methods should mimic the Python's `networkx` library. The floor division and modulo operations for negative numbers are preserved as per the Python's semantics.
