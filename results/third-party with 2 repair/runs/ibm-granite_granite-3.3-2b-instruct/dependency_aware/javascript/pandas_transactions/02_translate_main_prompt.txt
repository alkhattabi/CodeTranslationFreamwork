import pandas as pd
from utils import category_balance
from service import rolling_alert

def solve(n: int) -> int:
    balance = category_balance(n)
    alert = rolling_alert(n)
    scores = pd.Series([abs(balance) % 50, alert, abs(n) % 50])
    ranks = scores.rank(method="dense", ascending=False)
    penalty = int(ranks.iloc[0] * 7 + ranks.iloc[1] * 3)
    return balance + alert - penalty

export function rollingAlert(n) {
    n = Math.max(-200, Math.min(200, n));
    const series = [
        Math.floor(n % 13),
        Math.floor((n / 2) % 13),
        Math.floor(n % 17),
        Math.abs(Math.floor(n)),
        Math.floor(n % 7)
    ];

    const rolling = series.reduce((acc, curr, index) => {
        if (index === 0) return [acc, series[index + 1]];
        return [acc[acc.length - 1], acc[acc.length - 1] + curr];
    }, [series[0]]);

    const alerts = Math.floor((rolling.reduce((sum, curr) => sum + curr, 0) >= 15).length);
    return (alerts * 10) + Math.floor(categoryBalance(n));
}

function categoryBalance(n) {
    n = Math.max(-200, Math.min(200, n));
    const frame = [
        {"category": "A", "amount": n},
        {"category": "B", "amount": 12},
        {"category": "A", "amount": n / 2},
        {"category": "C", "amount": -7},
        {"category": "B", "amount": 9 % 10},
        {"category": "C", "amount": 4},
        {"category": "A", "amount": -n / 3}
    ];

    const sums = frame.reduce((acc, curr) => {
        acc[curr.category] = (acc[curr.category] || 0) + curr.amount;
        return acc;
    }, {});

    return Math.floor(2 * sums["A"] + sums["B"] - sums["C"]);
}
