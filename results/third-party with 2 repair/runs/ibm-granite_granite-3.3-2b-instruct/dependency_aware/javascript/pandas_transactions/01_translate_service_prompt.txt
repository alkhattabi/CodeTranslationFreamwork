def rolling_alert(n: int) -> int:
    n = max(-200, min(200, n))
    series = pd.Series([n % 13, (n // 2) % 13, n % 17, abs(n) % 11, n % 7])
    rolling = series.rolling(window=3, min_periods=1).sum()
    alerts = int((rolling >= 15).sum())

export function categoryBalance(n) {
    n = Math.max(-200, Math.min(200, n));
    const frame = [
        {"category": "A", "amount": n},
        {"category": "B", "amount": 12},
        {"category": "A", "amount": n / 2},
        {"category": "C", "amount": -7},
        {"category": "B", "amount": 9 % 10},
        {"category": "C", "amount": 4},
        {"category": "A", "amount": -n / 3}
    ];

    const sums = frame.reduce((acc, curr) => {
        acc[curr.category] = (acc[curr.category] || 0) + curr.amount;
        return acc;
    }, {});

    return Math.floor(2 * sums["A"] + sums["B"] - sums["C"]);
}
