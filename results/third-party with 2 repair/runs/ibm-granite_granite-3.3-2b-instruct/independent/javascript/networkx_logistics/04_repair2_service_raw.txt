import { route_cost } from "./utils.mjs";

function disruption_score(n) {
    const graph = {
        vertices: [0, 1, 2, 3, 4, 5, 6],
        edges: [
            [0, 1],
            [0, 2],
            [1, 3],
            [2, 3],
            [3, 4],
            [4, 5],
            [2, 5],
            [5, 6]
        ]
    };

    if (n % 3 === 0) {
        delete graph.edges[1]; // Remove edge (2, 5)
    }
    if (n % 5 === 0) {
        delete graph.edges[0]; // Remove edge (0, 2)
    }

    let reachable = 0;
    function dfs(node) {
        reachable++;
        for (const neighbor of graph.edges) {
            if (neighbor[0] === node) {
                dfs(neighbor[1]);
            }
        }
    }
    dfs(0);

    let longest = 0;
    function dfsPath(node, visited, path) {
        visited[node] = true;
        path.push(node);
        longest = Math.max(longest, path.length);

        for (const neighbor of graph.edges) {
            if (!visited[neighbor[1]]) {
                dfsPath(neighbor[1], visited, path);
            }
        }
    }
    const visited = new Array(graph.vertices.length).fill(false);
    dfsPath(0, visited, []);

    return route_cost(n) + 3 * reachable + longest;
}

export { disruption_score };
