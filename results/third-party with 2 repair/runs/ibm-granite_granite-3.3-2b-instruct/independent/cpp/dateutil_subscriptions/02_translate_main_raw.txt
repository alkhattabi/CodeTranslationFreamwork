#include "api.hpp"

int solve(int n) {
    int renewed = grace_period(n);
    int deadline = grace_period(n);
    if (renewed < 0 || deadline < 0) {
        return -1;
    }

    int y1 = renewed / 10000;
    int m1 = (renewed / 100) % 100;
    int d1 = renewed % 100;

    int y2 = deadline / 10000;
    int m2 = (deadline / 100) % 100;
    int d2 = deadline % 100;

    int days = (deadline - renewed) / 10000 - 1;
    int d2_ = deadline % 100;

    return days * 100 + d2_;
}
