// service.mjs
import { renewal_date } from "./utils.mjs";

export function grace_period(n) {
    const renewed = renewal_date(n);
    if (renewed < 0) {
        return -1;
    }
    const year = Math.floor(renewed / 10000);
    const month = Math.floor((renewed / 100) % 100);
    const day = renewed % 100;
    const due = new Date(year, month - 1, day).setMonth(month - 1 + 2, day + 7);
    return due.getFullYear() * 10000 + due.getMonth() * 100 + due.getDate();
}

import { date } from 'date-fns';

export function renewal_date(n) {
    if (n < 19000101 || n > 20991231) {
        return -1;
    }
    const year = Math.floor(n / 10000);
    const month = Math.floor((n / 100) % 100);
    const day = n % 100;

    try {
        const start = new Date(year, month - 1, day);
    } catch (error) {
        return -1;
    }

    const end = new Date(start.getFullYear(), start.getMonth() + 1, 0);

    return end.getFullYear() * 10000 + end.getMonth() * 100 + end.getDate();
}
