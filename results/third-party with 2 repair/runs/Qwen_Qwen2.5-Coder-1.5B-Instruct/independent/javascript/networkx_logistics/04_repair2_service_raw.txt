// service.mjs
export function disruption_score(n) {
    const graph = new Map();
    graph.set(0, [1, 2]);
    graph.set(1, [3]);
    graph.set(2, [3, 5]);
    graph.set(3, [4]);
    graph.set(4, [5]);
    graph.set(5, [6]);

    if (n % 3 === 0) {
        graph.delete(2);
    }
    if (n % 5 === 0) {
        graph.delete(0);
    }

    let reachable = 0;
    for (const node of graph.keys()) {
        reachable += graph.get(node).length;
    }

    const longest = Math.max(...graph.values().map(path => path.length));

    return routeCost(n) + 3 * reachable + parseInt(longest);
}
