        int[][] m = new int[][] {

        int[][] square = new int[3][3];

                square[i][j] = m[i][j] * m[j][i];
            }
        }
        int diagonal = 0;
        for (int i = 0; i < 3; i++) {
            diagonal += square[i][i];
        }
        return diagonal + vector_signature(n);
    }
}
