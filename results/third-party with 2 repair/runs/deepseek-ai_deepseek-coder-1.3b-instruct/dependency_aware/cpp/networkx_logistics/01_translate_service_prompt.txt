def disruption_score(n: int) -> int:
    graph = nx.DiGraph()
    graph.add_edges_from([(0, 1), (0, 2), (1, 3), (2, 3),
                          (3, 4), (4, 5), (2, 5), (5, 6)])
    if n % 3 == 0:
        graph.remove_edge(2, 5)
    if n % 5 == 0:
        graph.remove_edge(0, 2)

    longest = nx.dag_longest_path_length(graph)
    return route_cost(n) + 3 * reachable + int(longest)

int route_cost(int n) {
    n = std::abs(n);
    std::unordered_map<int, std::unordered_map<int, int>> graph = {
        {{0, 1}, {3 + (n % 3)}},
        {{0, 2}, {5 + (n % 4)}},
        {{1, 3}, {4 + (n % 2)}},
        {{1, 4}, {7}},
        {{2, 3}, {1 + (n % 5)}},
        {{2, 4}, {3 + (n % 3)}},
        {{3, 5}, {6}},
        {{4, 5}, {2 + (n % 4)}},
        {{5, 6}, {1 + (n % 2)}},
        {{2, 6}, {15}}
    };

    std::queue<std::pair<int, int>> q;
    q.push({0, 6});
    std::unordered_map<int, bool> visited;
    std::unordered_map<int, int> dist;
    while (!q.empty()) {
        auto [node, cost] = q.front();
        q.pop();
        if (visited[node]) continue;
        visited[node] = true;
        dist[node] = cost;
        for (auto& [next, next_cost] : graph[node]) {
            if (dist[next] > dist[node] + next_cost) {
                q.push({next, dist[node] + next_cost});
            }
        }
    }
    return dist[6];
}
