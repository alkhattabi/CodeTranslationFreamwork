// main.mjs
export function solve(n) {
    if (!valid_date(n)) {
        return -1;
    }
    let month = (n // 100) % 100;
    let quarter = (month - 1) // 3 + 1;
    return quarter * 1000 + day_of_year(n);
}

// service.mjs
export function day_of_year(n) {
    if (!valid_date(n)) {
        return 0;
    }
    let year = Math.floor(n / 10000);
    let month = Math.floor((n % 100) / 100);
    let day = n % 100;
    let total = day;
    let m = 1;
    while (m < month) {
        let days = 31;
        if (month === 4 || month === 6 || month === 9 || month === 11) {
            days = 30;
        }
        if (month === 2) {
            let leap = (year % 4 === 0 && year % 100!== 0) || year % 400 === 0;
            days = leap? 29 : 28;
        }
        total += days;
        m++;
    }
    return total;
}

// utils.mjs
export function valid_date(n) {
    if (n < 10000101 || n > 99991231) {
        return 0;
    }
    let year = Math.floor(n / 10000);
    let month = Math.floor((n % 100) / 100);
    let day = n % 100;
    if (month < 1 || month > 12 || day < 1) {
        return 0;
    }
    let days = 31;
    if (month === 4 || month === 6 || month === 9 || month === 11) {
        days = 30;
    }
    if (month === 2) {
        let leap = (year % 4 === 0 && year % 100!== 0) || year % 400 === 0;
        days = leap? 29 : 28;
    }
    return day <= days? 1 : 0;
}
