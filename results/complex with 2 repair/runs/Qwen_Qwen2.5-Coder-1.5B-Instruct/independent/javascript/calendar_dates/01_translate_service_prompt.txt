from utils import valid_date

def day_of_year(n: int) -> int:
    if valid_date(n) == 0:
        return 0
    year = n // 10000
    month = (n // 100) % 100
    day = n % 100
    total = day
    m = 1
    while m < month:
        days = 31
        if m == 4 or m == 6 or m == 9 or m == 11:
            days = 30
        if m == 2:
            leap = (year % 4 == 0 and year % 100 != 0) or year % 400 == 0
            days = 29 if leap else 28
        total += days
        m += 1
    return total
