            return 4;

public class Service {
    public static int yearly_interest(int n) {
        int band = Utils.credit_band(n);
        if (band == 0) {
            return -1;
        }
        int principal = 10000 + (n - 300) * 10;
        int rate = 18 - band * 3;
        int interest = (int) (principal * rate / 100.0);
        if (principal > 14000) {
            interest += 125;
        }
        return interest;
    }
}

public class Main {
    public static int solve(int n) {
        int band = Utils.credit_band(n);
        if (band == 0) {
            return -1;
        }
        int principal = 10000 + (n - 300) * 10;
        int interest = Service.yearly_interest(n);
        int fee = 150;
        if (band >= 3) {
            fee = 350;
        }
        int total = principal + interest + fee;
        return (total + 11) / 12;
    }
}
