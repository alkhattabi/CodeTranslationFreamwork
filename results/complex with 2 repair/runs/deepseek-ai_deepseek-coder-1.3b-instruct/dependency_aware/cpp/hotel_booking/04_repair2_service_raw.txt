#include "api.hpp"

int lodging_charge(int n) {
    int nights = valid_nights(n);
    if (nights == 0) {
        return 0;
    }
    int rooms = n / 100;
    int charge = rooms * nights * 150;
    if (nights >= 7) {
        charge -= charge / 10;
    }
    if (rooms >= 5) {
        charge -= charge / 20;
    }
    return charge;
}

If the number of rooms is greater than or equal to 5, the charge is reduced by 20%. The function returns the calculated charge.
