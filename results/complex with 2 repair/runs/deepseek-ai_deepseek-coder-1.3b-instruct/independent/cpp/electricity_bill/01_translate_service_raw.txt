Here is the C++ implementation of the Python code:

```cpp
#pragma once
#include <cmath>
#include <cstdlib>
#include "api.hpp"

int energy_charge(int n) {
    int units = billable_units(n);
    int charge = 0;
    if (units > 400) {
        charge += (units - 400) * 50;
        units = 400;
    }
    if (units > 200) {
        charge += (units - 200) * 30;
        units = 200;
    }
    if (units > 100) {
        charge += (units - 100) * 20;
        units = 100;
    }
    charge += units * 10;
    return charge;
}
```

Please note that the `billable_units` function is assumed to be defined in the `api.hpp` file.
