def luhn_valid(n: int) -> int:
    if n <= 0:
        return 0
    total = 0
    double = False
    while n > 0:
        digit = n % 10
        if double:
            digit *= 2
            if digit > 9:
                digit -= 9
        total += digit
        double = not double
        n //= 10
