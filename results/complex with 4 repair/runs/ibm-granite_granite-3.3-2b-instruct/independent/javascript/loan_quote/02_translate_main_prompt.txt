from utils import credit_band
from service import yearly_interest

def solve(n: int) -> int:
    band = credit_band(n)
    if band == 0:
        return -1
    principal = 10000 + (n - 300) * 10
    interest = yearly_interest(n)
    fee = 150 if band >= 3 else 350
