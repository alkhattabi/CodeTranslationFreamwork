import { billableUnits } from "./utils.mjs";

export function energyCharge(n) {
    let units = billableUnits(n);
    let charge = 0;
    if (units > 400) {
        charge += (units - 400) * 50;
        units = 400;
    }
    if (units > 200) {
        charge += (units - 200) * 30;
        units = 200;
    }
    if (units > 100) {
        charge += (units - 100) * 20;
        units = 100;
    }
    charge += units * 10;
    return charge;
}
