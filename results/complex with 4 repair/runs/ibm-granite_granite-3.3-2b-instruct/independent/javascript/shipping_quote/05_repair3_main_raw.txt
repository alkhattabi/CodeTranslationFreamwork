export function solve(n) {
    const blocks = weight_blocks(n);
    if (blocks === 0) {
        return 0;
    }
    const cost = transport_cost(n);
    const fuel = (cost * 8 + 99) / 100;
    const insurance = blocks >= 10? 300 : 100;
    return cost + fuel + insurance;
}

export function transport_cost(n) {
    const blocks = weight_blocks(n);
    if (blocks === 0) {
        return 0;
    }
    let price = 1200 + blocks * 180;
    if (blocks > 5) {
        price += (blocks - 5) * 70;
    }
    if (blocks > 12) {
        price += (blocks - 12) * 90;
    }
    return price;
}

export function weight_blocks(n) {
    if (n <= 0) {
        return 0;
    }
    if (n > 10000) {
        n = 10000;
    }
    return (n + 499) / 500;
}
