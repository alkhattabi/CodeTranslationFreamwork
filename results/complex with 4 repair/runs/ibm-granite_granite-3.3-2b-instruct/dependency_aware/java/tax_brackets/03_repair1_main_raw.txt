public class Main {
    public static int solve(int n) {
        int base = Utils.taxable_income(n);
        int tax = Service.calculate_tax(n);
        int credit = 100;
        if (base > 0 && base <= 2500) {
            credit = 0;
        }
        int payable = tax - credit;
        if (payable < 0) {
            payable = 0;
        }
        payable += (payable * 2 / 100) / 100; // 2% of payable if base > 10000
        return payable;
    }
}
