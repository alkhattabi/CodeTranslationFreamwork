        if (n <= 0) {
            return 0;
        }
        int total = 0;
        boolean doubled = false;
        while (n > 0) {
            int digit = n % 10;
            if (doubled) {
                digit *= 2;
                if (digit > 9) {
                    digit -= 9;
                }

            doubled =!doubled;

    }
