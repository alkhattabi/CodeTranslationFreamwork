// service.cpp
#include "api.hpp"

int transport_cost(int n) {
    int blocks = solve(n);
    if (blocks == 0) {
        return 0;
    }
    int price = 1200 + blocks * 180;
    if (blocks > 5) {
        price += (blocks - 5) * 70;
    }
    if (blocks > 12) {
        price += (blocks - 12) * 90;
    }
    return price;
}

#include "api.hpp"

int weight_blocks(int n) {
    if (n <= 0) {
        return 0;
    }
    if (n > 10000) {
        n = 10000;
    }
    return (n + 499) / 500;
}
