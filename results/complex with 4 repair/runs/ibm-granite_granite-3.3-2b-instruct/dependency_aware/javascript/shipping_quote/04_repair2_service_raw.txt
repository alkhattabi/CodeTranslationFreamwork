// service.mjs
export function transportCost(n) {
    const blocks = weightBlocks(n);
    if (blocks === 0) {
        return 0;
    }
    let price = 1200 + blocks * 180;
    if (blocks > 5) {
        price += (blocks - 5) * 70;
    }
    if (blocks > 12) {
        price += (blocks - 12) * 90;
    }
    return price;
}

// utils.mjs (no changes needed)
export function weightBlocks(n) {
    if (n <= 0) {
        return 0;
    }
    if (n > 10000) {
        n = 10000;
    }
    return (n + 499) / 500;
}
