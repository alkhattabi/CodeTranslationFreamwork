```javascript
import { weeklyDemand } from "./utils.mjs";

export function order_quantity(n) {
    const demand = weeklyDemand(n);
    const safety = 15 < demand ? 15 : 35;
    const available = n > 0 ? n : 0;
    const shortage = demand + safety - available;
    if (shortage <= 0) {
        return 0;
    }
    const packs = Math.ceil((shortage + 11) / 12);
    return packs * 12;
}
```
