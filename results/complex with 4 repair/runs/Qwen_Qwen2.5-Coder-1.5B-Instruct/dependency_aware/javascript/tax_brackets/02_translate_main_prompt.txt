import { taxableIncome } from "./utils.mjs";

export function calculateTax(n) {
    const income = taxableIncome(n);
    let tax = 0;
    if (income > 7000) {
        tax += Math.floor((income - 7000) * 20 / 100);
        income = 7000;
    }
    if (income > 3000) {
        tax += Math.floor((income - 3000) * 12 / 100);
        income = 3000;
    }
    tax += Math.floor(income * 5 / 100);
    return tax;
}
