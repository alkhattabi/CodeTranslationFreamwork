import java.util.function.IntPredicate;

public class Service {

    public static int lodging_charge(int n) {
        IntPredicate isValidNights = n -> n > 0 && n <= 9999;
        if (!isValidNights.test(n)) {
            throw new IllegalArgumentException("Invalid number of nights");
        }

        int nights = Utils.valid_nights(n);
        if (nights == 0) {
            return 0;
        }

        int rooms = n / 100;
        int charge = rooms * nights * 150;
        if (nights >= 7) {
            charge -= charge / 10;
        }
        if (rooms >= 5) {
            charge -= charge / 20;
        }
        return charge;
    }
}

public class Utils {
    public static int valid_nights(int n) {
        if (n <= 0) {
            return 0;
        }
        int rooms = n / 100;
        int nights = n % 100;
        if (rooms < 1 || rooms > 20 || nights < 1 || nights > 30) {
            return 0;
        }
        return nights;
    }
}
