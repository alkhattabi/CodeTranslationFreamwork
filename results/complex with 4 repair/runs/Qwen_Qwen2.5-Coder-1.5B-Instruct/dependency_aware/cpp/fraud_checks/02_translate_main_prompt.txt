int luhn_valid(int n) {
    if (n <= 0) {
        return 0;
    }
    int total = 0;
    bool double_ = false;
    while (n > 0) {
        int digit = n % 10;
        if (double_) {
            digit *= 2;
            if (digit > 9) {
                digit -= 9;
            }
        }
        total += digit;
        double_ = !double_;
        n /= 10;
    }
    return 1 if total % 10 == 0 else 0;
}
