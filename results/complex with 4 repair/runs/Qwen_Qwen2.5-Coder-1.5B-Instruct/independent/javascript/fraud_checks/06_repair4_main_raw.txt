```javascript
import { riskPoints } from './service.mjs';
import { luhnValid } from './utils.mjs';

export function solve(n: number): number {
    const points = riskPoints(n);
    if (points >= 70) {
        return 3;
    }
    if (points >= 40) {
        return 2;
    }
    if (luhnValid(n) === 1) {
        return 0;
    }
    return 1;
}
```
