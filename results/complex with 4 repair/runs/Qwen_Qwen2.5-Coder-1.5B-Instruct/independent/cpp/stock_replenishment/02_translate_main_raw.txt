```cpp
#pragma once
#include <algorithm>
#include <cmath>
#include <cstdlib>

// Helper function declarations from api.hpp
int order_quantity(int n);
int weekly_demand(int n);

int solve(int n) {
    int quantity = order_quantity(n);
    if (quantity == 0) {
        return 0;
    }
    int demand = weekly_demand(n);
    int unit_cost = 25;
    if (quantity >= 120) {
        unit_cost = 30;
    }
    int shipping = 400;
    if (demand > 100) {
        shipping = 200;
    }
    return quantity * unit_cost + shipping;
}
```
