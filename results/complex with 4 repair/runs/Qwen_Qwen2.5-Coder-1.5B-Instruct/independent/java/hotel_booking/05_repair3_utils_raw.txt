package com.example;

public class Utils {
    public static int valid_nights(int n) {
        if (n <= 0) {
            return 0;
        }
        int rooms = n / 100;
        int nights = n % 100;
        if (rooms < 1 || rooms > 20 || nights < 1 || nights > 30) {
            return 0;
        }
        return nights;
    }
}
