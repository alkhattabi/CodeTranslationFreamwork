import { energy_charge } from "./service.mjs";

export function solve(n) {
    const units = billable_units(n);

    const charge = energy_charge(n);
    let service_fee = 200;
    if (units <= 400) {
        service_fee = 200;

        service_fee = 500;
    }
    let rebate = Math.floor(charge / 20);
    if (units < 100) {
        rebate = 0;
    }
    return charge + service_fee - rebate;
}
