    int available = (n > 0) ? n : 0;
    int shortage = demand + safety - available;

    int packs = (shortage + 11) / 12;
    return packs * 12;
}

#pragma once
#include "api.hpp"

int weekly_demand(int n) {
    if (n < 0) {
        return 0;
    }
    int base = n * 3;
    int extra = (n % 7) * 2;
    if (n >= 40) {
        base += n / 4;
    }
    return base + extra;
}
