export function yearly_interest(n) {
    const band = credit_band(n);
    if (band === 0) {
        return -1;
    }
    const principal = 10000 + (n - 300) * 10;
    const rate = 18 - band * 3;
    const interest = principal * rate / 100;
    if (principal > 14000) {
        interest += 125;
    }
    return interest;
}
