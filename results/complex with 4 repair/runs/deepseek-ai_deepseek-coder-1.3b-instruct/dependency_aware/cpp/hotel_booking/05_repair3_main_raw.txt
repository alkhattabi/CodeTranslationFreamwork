```cpp
#include "api.hpp"

int solve(int n) {
    int nights = valid_nights(n);
    if (nights == 0) {
        return -1;
    }
    int rooms = n / 100;
    int charge = lodging_charge(n);
    int cleaning = rooms * 45;
    int tax = (charge + cleaning) * 15 / 100;
    return charge + cleaning + tax;
}

int lodging_charge(int n) {
    int nights = valid_nights(n);
    if (nights == 0) {
        return 0;
    }
    int rooms = n / 100;
    int charge = rooms * nights * 150;
    if (nights >= 7) {
        charge -= charge / 10;
    }
    if (rooms >= 5) {
        charge -= charge / 20;
    }
    return charge;
}

int valid_nights(int n) {
    if (n <= 0) {
        return 0;
    }
    int rooms = n / 100;
    int nights = n % 100;
    if (rooms < 1 || rooms > 20 || nights < 1 || nights > 30) {
        return 0;
    }
    return nights;
}
```
